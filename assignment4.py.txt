# Create two matrices
A = [[1, 2], [3, 4]]
B = [[5, 6], [7, 8]]

# Add two matrices
C = [[0, 0], [0, 0]]

for i in range(2):
    for j in range(2):
        C[i][j] = A[i][j] + B[i][j]

print("Matrix A:", A)
print("Matrix B:", B)
print("Addition of matrices:")
for row in C:
    print(row)
